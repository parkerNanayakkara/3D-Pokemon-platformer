// Check Encounter Grass Zones let inZone = false; for (let z of this.encounterZones) { if (pos.x >= z.xMin && pos.x <= z.xMax && pos.z >= z.zMin && pos.z <= z.zMax) { this.updateLocation(z.regionName); inZone = true; break; } } if (!inZone) { if (pos.x > 300) this.updateLocation("Alola Region"); else if (pos.x < -300) this.updateLocation("Galar Region"); else this.updateLocation("Pallet Town"); } this.trySpawnWildPokemon(); // Wild Pokémon world interaction for (let i = this.spawnedWilds.length - 1; i >= 0; i--) { const wild = this.spawnedWilds[i]; const progress = Math.min((time - wild.startTime) / wild.duration, 1.0); const easeOut = Math.sin((progress * Math.PI) / 2); if (!wild.active) { wild.mesh.position.y = -2 + (wild.targetY + 2) * easeOut; if (progress >= 1.0) wild.active = true; } else { wild.mesh.rotation.y += 0.02; wild.mesh.position.y = 0.4 + Math.sin(time * 0.005) * 0.08; } const distToPlayer = wild.mesh.position.distanceTo(this.camera.position); if (wild.active && distToPlayer < 2.5 && this.state === 'EXPLORE') { this.triggerEncounter([wild.species], wild); break; } }
